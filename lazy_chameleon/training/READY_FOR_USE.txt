   from synthetic_data_generator import SyntheticDataGenerator, DatasetExporter
   from anthropic import Anthropic
   
   client = Anthropic(api_key="sk-...")
   def teacher_client(prompt, max_tokens, temp):
       msg = client.messages.create(
           model="claude-opus",
           max_tokens=max_tokens,
           temperature=temp,
           messages=[{"role": "user", "content": prompt}]
       )
       return msg.content[0].text
   
   generator = SyntheticDataGenerator(teacher_client, {
       "max_tokens": 2000,
       "temperature": 0.8,
       "teacher_model": "claude-opus"
   })

3. GENERATE & EXPORT:
   data = generator.generate_batch(n=1000)
   DatasetExporter.to_chatml(data, "training.jsonl")
